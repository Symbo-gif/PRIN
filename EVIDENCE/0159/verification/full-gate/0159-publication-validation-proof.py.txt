def pdf_fixture(
    creator: bytes,
    producer: bytes,
    date: bytes,
    identifier: bytes,
    label: bytes = b"SCIENCE",
    title: bytes = b"unchanged",
) -> bytes:
    """Build a classic-xref PDF with a real page, stream, Info and font."""
    stream = (
        b"/Artifact <</Creator (SCIENTIFIC PROPERTY)>> BDC\n"
        b"BT /F1 8 Tf 2 2 Td (" + label + b") Tj ET\nEMC\n"
    )
    objects = [
        b"<< /Type /Catalog /Pages 2 0 R >>",
        b"<< /Type /Pages /Count 1 /Kids [3 0 R] >>",
        b"<< /Type /Page /Parent 2 0 R /MediaBox [0 0 10 10] "
        b"/Contents 4 0 R /Resources << /Font << /F1 6 0 R >> >> >>",
        b"<< /Length " + str(len(stream)).encode() + b" >>\nstream\n"
        + stream + b"endstream",
        b"<< /Creator (" + creator + b") /Producer (" + producer
        + b") /CreationDate (" + date + b") /ModDate (" + date
        + b") /Title (" + title + b") >>",
        b"<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica >>",
    ]
    payload = bytearray(b"%PDF-1.4\n%\xff\xfe\n")
    offsets = []
    for number, value in enumerate(objects, 1):
        offsets.append(len(payload))
        payload += f"{number} 0 obj\n".encode() + value + b"\nendobj\n"
    xref = len(payload)
    payload += b"xref\n0 7\n0000000000 65535 f \n"
    for offset in offsets:
        payload += f"{offset:010d} 00000 n \n".encode()
    payload += (
        b"trailer\n<< /Size 7 /Root 1 0 R /Info 5 0 R /ID [<"
        + identifier + b"> <" + identifier + b">] >>\nstartxref\n"
        + str(xref).encode() + b"\n%%EOF\n"
    )
    return bytes(payload)


def png_fixture(chunks: list[tuple[bytes, bytes]]) -> bytes:
    """Build CRC-correct PNG chunks around a real one-pixel RGB image."""
    payload = bytearray(b"\x89PNG\r\n\x1a\n")
    for kind, data in chunks:
        payload += (
            struct.pack(">I", len(data)) + kind + data
            + struct.pack(">I", zlib.crc32(kind + data))
        )
    return bytes(payload)


def rejects(function: object, payload: bytes) -> None:
    """An invalid container must fail as a DriverError, not a mismatch."""
    try:
        function(payload)
    except DriverError:
        return
    raise AssertionError("malformed publication was admitted")
